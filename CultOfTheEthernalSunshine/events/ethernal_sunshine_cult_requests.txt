namespace = ethernal_sunshine_cult_requests

#Base event that extracts the possible cult requests
country_event = {
	id = ethernal_sunshine_cult_requests.0
	hide_window = yes
	is_triggered_only = yes

	trigger = {
        has_origin = origin_ethernal_sunshine_cult
    }

    immediate = {
        random_list = {
			1 = {
				country_event = {
					id = ethernal_sunshine_cult_requests.101
				}
			}
			1 = {
				country_event = {
					id = ethernal_sunshine_cult_requests.102
				}
			}
		}

		if = {
			limit = {
				has_country_flag = es_cult_established
			}
			country_event = {
				id = ethernal_sunshine_cult_requests.0
				days = 1800
				random = 3600
			}
		}
    }
}

country_event = {
	id = ethernal_sunshine_cult_requests.101
	hide_window = no
	is_triggered_only = yes

	title = es_req.1.title
	desc = es_req.1.desc

	trigger = {
		has_origin = origin_ethernal_sunshine_cult
	}

	option = {
		name = es_req.101.a
		allow = {
			resource_stockpile_compare = {
				resource = food
				value >= 1000
			}
		}
		add_resource = { food = -1000 }
	}
	option = {
		name = es_req.101.b
		add_resource = {
			unity = -500
		}
		custom_tooltip = cult_disappointment_modifier.tooltip
		hidden_effect = {
			country_event = {
				id = ethernal_sunshine_cult_requests.1000
			}
		}
	}
}

	hide_window = yes
	is_triggered_only = yes

	immediate = {
		every_pop_faction = {
			if = {
				limit = {
					has_pop_faction_flag = es_cult
				}
				add_modifier = {
					#modifier = cult_disappointment_modifier
					modifier = cult_disappointment_modifier
					days = 720
				}
			}
		}
	}
